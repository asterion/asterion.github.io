<?php

declare(strict_types=1);

namespace Asterion\Developer;

/**
 * Desarrollador web desde 2006.
 * Cree en el poder de Internet y en
 * construir software simple, mantenible
 * y útil.
 */
#[Hireable]
final readonly class Marcos
	implements WebDeveloper
{
	public function __construct(
		public string $name = 'Marcos',
		public string $role = 'Web Developer',
		public string $github = '@asterion',
	) {}

	public function backend(): array
	{
		return ['PHP', 'Symfony', 'Laravel'];
	}

	public function frontend(): array
	{
		return ['HTML', 'CSS', 'JavaScript'];
	}

	public function databases(): array
	{
		return ['MySQL', 'PostgreSQL'];
	}

	public function infrastructure(): array
	{
		return [
			'GNU/Linux', 'Apache', 'Nginx',
			'Bash', 'Vim', 'Git',
		];
	}

	public function strengths(): array
	{
		return [
			'E-commerce a gran escala',
			'Sistemas CMS empresariales',
			'Calidad de código y testing',
			'Debugging y resolución de problemas',
		];
	}

	public function yearsOfExperience(): int
	{
		// 20 años y contando
		return (int) date('Y') - 2006;
	}

	public function hire(Project $p): Result
	{
		return $p
			|> $this->understand(...)
			|> $this->build(...)
			|> $this->ship(...);
	}
}

echo "¡Hola! Gracias por pasar.";
